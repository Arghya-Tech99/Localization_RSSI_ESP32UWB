/**
 * ============================================================================
 *  ESP32-DW1000 UWB Data Acquisition & Streaming Firmware
 * ============================================================================
 *
 *  Description:
 *    1. DW1000 transceiver receives UWB frames and computes diagnostic
 *       registers (RX_POWER, FP_POWER, etc.).
 *    2. ESP32 reads these registers via SPI and converts them to dBm.
 *    3. The firmware also extracts the Time of Arrival (TOA) timestamp.
 *    4. A structured payload is assembled and streamed over USB serial.
 *
 *  Hardware Connections (ESP32-DW1000 Evaluation Board):
 *    - SCK  → GPIO 18
 *    - MISO → GPIO 19
 *    - MOSI → GPIO 23
 *    - CS   → GPIO 5   (SPI chip select)
 *    - IRQ  → GPIO 34  (interrupt, pulled high when diagnostics ready)
 *    - RST  → GPIO 27  (reset, active low)
 *
 *  Reference: https://robu.in/product/esp32-dw1000-evaluation-board/
 * ============================================================================
 */

#include <Arduino.h>
#include <SPI.h>
#include "DW1000Ranging.h"
#include "DW1000.h"

// ============================================================================
//  PIN CONFIGURATION  (adjust to your board)
// ============================================================================

/** SPI pins – using the VSPI bus of the ESP32 */
#define PIN_SPI_SCK   18   /**< Serial Clock */
#define PIN_SPI_MISO  19   /**< Master In Slave Out */
#define PIN_SPI_MOSI  23   /**< Master Out Slave In */
#define PIN_SPI_SS    5    /**< Slave Select (Chip Select, active low) */

/** Control pins */
#define PIN_RST       27   /**< DW1000 reset (active low) */
#define PIN_IRQ       34   /**< DW1000 interrupt – rises when frame diags ready */

// ============================================================================
//  NODE IDENTITY  (must be unique per device)
// ============================================================================

/**
 * 64‑bit extended address (8 bytes, colon‑separated hex).
 * The first byte often indicates role:
 *   - 0x7D or 0x7F for TAG mode
 *   - 0x82 for ANCHOR mode
 *
 *  Change this address for each node in your network.
 */
static const char* NODE_ADDRESS = "7D:00:22:EA:82:60:3B:9C";

/**
 * Node role: "TAG" (mobile, initiates ranging) or "ANCHOR" (fixed, responds).
 * This firmware is written for a TAG that collects diagnostic data.
 */
static const char* NODE_ROLE = "TAG";

// ============================================================================
//  DW1000 OPERATING MODE  (trade‑off between range and speed)
// ============================================================================

/**
 * Available modes (see DW1000 library):
 *   - MODE_LONGDATA_RANGE_LOWPOWER : ~50 m range, low power consumption
 *   - MODE_SHORTDATA_FAST_LOWPOWER : shorter range but faster
 *   - MODE_LONGDATA_FAST_LOWPOWER  : longer range and faster (more power)
 *
 * Choose based on your environment:
 *   - indoor (<20 m)      : use SHORTDATA_FAST_LOWPOWER
 *   - indoor (20‑50 m)    : use LONGDATA_RANGE_LOWPOWER
 *   - outdoor (>50 m)     : may need HIGH_POWER (not shown)
 */
static constexpr DW1000::DeviceMode DW_MODE =
    DW1000::MODE_LONGDATA_RANGE_LOWPOWER;

// ============================================================================
//  ANTENNA DELAY CALIBRATION
// ============================================================================

/**
 * Antenna delay (in DW1000 clock cycles, ~15.65 ps per cycle).
 * This value compensates for internal and antenna propagation delays.
 * It must be calibrated for your specific hardware.
 *
 * Typical starting value for the ESP32‑DW1000 board is ~16384.
 * Calibration procedure: place two nodes at a known distance,
 * adjust this value until measured distance matches actual.
 */
static constexpr uint16_t ANTENNA_DELAY = 16384;

// ============================================================================
//  DATA STREAMING CONFIGURATION
// ============================================================================

/** Serial baud rate – must match monitor_speed in platformio.ini */
static constexpr uint32_t SERIAL_BAUD = 115200;

/** Delimiter for payload fields (CSV‑like format) */
static constexpr char DELIMITER = ',';

// ============================================================================
//  GLOBAL VARIABLES
// ============================================================================

/** Message buffer for building the output payload */
char payloadBuffer[256];

/** Indicates whether the interrupt has fired (diagnostics ready) */
volatile bool irqTriggered = false;

// ============================================================================
//  INTERRUPT SERVICE ROUTINE
// ============================================================================

/**
 * ISR for the DW1000 interrupt pin.
 * Simply sets a flag – heavy processing is done in the main loop.
 */
void IRAM_ATTR dw1000InterruptHandler() {
    irqTriggered = true;
}

// ============================================================================
//  HELPER FUNCTION: Convert register values to dBm
// ============================================================================

/**
 * Convert the raw RX_POWER register (from DW1000) to dBm.
 *
 * The formula is described in the DW1000 User Manual, section 7.2.2.
 *
 * @param rawPower   Raw 32‑bit value read from register 0x14 (RX_FINFO) or
 *                   from the diagnostic registers.
 * @return           Power in dBm (floating‑point).
 */
float rawPowerToDbm(uint32_t rawPower) {
    // Extract the 8‑bit exponent and 20‑bit mantissa
    int8_t exponent = (int8_t)((rawPower >> 24) & 0xFF);
    uint32_t mantissa = rawPower & 0x00FFFFFF;

    // Convert to linear power (scaled)
    double linear = (double)mantissa * pow(2.0, exponent) * 1e-6;

    // Convert to dBm: 10 * log10(linear) + 10*log10(1e-3) ?
    // But the DW1000 uses a reference of 1e-6 W, so the formula is:
    // dBm = 10 * log10( (mantissa * 2^exponent) / (2^31) ) + 10*log10(1e-3)
    // A simpler approach used by the library:
    // dBm = 10.0 * log10( (mantissa * pow(2.0, exponent)) / 0x7FFFFFFF ) - 30.0
    // We'll use the library's built‑in conversion for consistency.
    return DW1000::getPowerDbm(rawPower);
}

// ============================================================================
//  SETUP
// ============================================================================

void setup() {
    // ------------------------------------------------------------------------
    // 1. Serial communication
    // ------------------------------------------------------------------------
    Serial.begin(SERIAL_BAUD);
    while (!Serial) { ; }  // wait for USB serial (optional)

    Serial.println("=========================================");
    Serial.println("ESP32-DW1000 UWB Data Acquisition");
    Serial.println("=========================================");
    Serial.printf("Node Address: %s\n", NODE_ADDRESS);
    Serial.printf("Node Role   : %s\n", NODE_ROLE);
    Serial.printf("Mode        : %d\n", (int)DW_MODE);
    Serial.printf("Antenna delay: %d\n", ANTENNA_DELAY);
    Serial.println("-----------------------------------------");

    // ------------------------------------------------------------------------
    // 2. SPI initialization
    // ------------------------------------------------------------------------
    // The DW1000 library expects SPI to be already set up.
    // We use the default VSPI pins (SCK=18, MISO=19, MOSI=23, SS=5).
    // If you need custom pins, use SPI.begin(PIN_SPI_SCK, PIN_SPI_MISO, PIN_SPI_MOSI, PIN_SPI_SS)
    SPI.begin(PIN_SPI_SCK, PIN_SPI_MISO, PIN_SPI_MOSI, PIN_SPI_SS);

    // ------------------------------------------------------------------------
    // 3. DW1000 initialization
    // ------------------------------------------------------------------------
    // Set the chip select pin
    DW1000Ranging.initCommunication(PIN_SPI_SS, PIN_RST);

    // Set the device mode and antenna delay (both are configurable)
    DW1000Ranging.attachDeviceMode(DW_MODE);
    DW1000Ranging.setAntennaDelay(ANTENNA_DELAY);

    // Set the node address
    DW1000Ranging.setDeviceAddress(NODE_ADDRESS);
    DW1000Ranging.setDeviceRole(NODE_ROLE);

    // Start ranging (this also configures the transceiver)
    DW1000Ranging.startRanging();

    // ------------------------------------------------------------------------
    // 4. Interrupt setup
    // ------------------------------------------------------------------------
    // Configure the IRQ pin as input with pull‑up (the DW1000 drives it high)
    pinMode(PIN_IRQ, INPUT_PULLUP);
    // Attach the interrupt on RISING edge (when diagnostics are ready)
    attachInterrupt(digitalPinToInterrupt(PIN_IRQ), dw1000InterruptHandler, RISING);

    Serial.println("DW1000 initialized. Waiting for incoming frames...");
}

// ============================================================================
//  MAIN LOOP
// ============================================================================

void loop() {
    // ------------------------------------------------------------------------
    // 1. Check if an interrupt has occurred (diagnostics ready)
    // ------------------------------------------------------------------------
    if (irqTriggered) {
        // Reset the flag immediately to avoid missing subsequent interrupts
        irqTriggered = false;

        // --------------------------------------------------------------------
        // 2. Read diagnostic registers from DW1000
        // --------------------------------------------------------------------
        // The library provides methods to read the latest ranging data.
        // However, for diagnostic values, we need to access the internal
        // registers directly. The library exposes raw register read/write.

        // We'll read the RX_FINFO register (0x14) which contains:
        //   - RX_POWER (total received power)
        //   - FP_INDEX (first path index) etc.
        // But the FP_POWER is available via the "RX_TIME" and "RX_POWER" registers.
        // Alternatively, we can use the library's getFirstPathPower().

        // For clarity, we retrieve values using the DW1000 class methods.
        uint32_t rawRxPower = DW1000::readRegister(DW1000::REG_RX_FINFO, 4);
        uint32_t rawFirstPathPower = DW1000::readRegister(DW1000::REG_RX_FINFO, 4); // actual register is different; use library function

        // The library provides a more convenient method:
        // Get RSSI (in dBm) and First Path Power (in dBm)
        float rssiDbm = DW1000::getPowerDbm(rawRxPower);
        float fpPowerDbm = DW1000::getFirstPathPower();

        // --------------------------------------------------------------------
        // 3. Get Time of Arrival (TOA) timestamp
        // --------------------------------------------------------------------
        // The DW1000 provides a 40‑bit timestamp (in device clock units ~15.65 ps)
        // that represents the time when the first path arrived.
        // We read it from the RX_TIME register (0x15) – but the library has a helper.
        uint64_t rxTimestamp = DW1000::getRxTimestamp(); // 40‑bit value

        // Convert to microseconds for easier reading (optional)
        // The clock frequency is ~63.8976 GHz, so period = 1/63.8976e9 ≈ 15.65e-12 s
        // To get microseconds: timestamp * 15.65e-12 * 1e6 = timestamp * 1.565e-5
        double toaUs = (double)rxTimestamp * 1.565e-5;

        // --------------------------------------------------------------------
        // 4. Assemble the structured payload
        // --------------------------------------------------------------------
        // Payload format: [Timestamp, Tx_Node, Rx_Node, RSSI_dBm, FP_Power_dBm, TOA_us]
        // - Timestamp: millis() from ESP32 (or system time)
        // - Tx_Node: the transmitter address (we get it from the received frame)
        // - Rx_Node: this node's address (NODE_ADDRESS)
        // - RSSI_dBm: total received signal power in dBm
        // - FP_Power_dBm: first path power in dBm
        // - TOA_us: Time of Arrival in microseconds

        // Get the transmitter address from the last received frame
        char txAddress[18];
        DW1000Ranging.getRemoteDeviceAddress(txAddress, 18); // copies the address string

        // Build the payload string
        snprintf(payloadBuffer, sizeof(payloadBuffer),
                 "%lu%c%s%c%s%c%.2f%c%.2f%c%.3f",
                 millis(),                         // ESP32 timestamp (ms)
                 DELIMITER,
                 txAddress,                        // Transmitter node address
                 DELIMITER,
                 NODE_ADDRESS,                     // Receiver (this node)
                 DELIMITER,
                 rssiDbm,                          // RSSI [dBm]
                 DELIMITER,
                 fpPowerDbm,                       // First path power [dBm]
                 DELIMITER,
                 toaUs);                           // Time of Arrival [µs]

        // --------------------------------------------------------------------
        // 5. Stream the payload over Serial
        // --------------------------------------------------------------------
        Serial.println(payloadBuffer);

        // (Optional) you could also transmit via Wi-Fi or save to SD card.
        // For this example, we just send over USB serial.

        // --------------------------------------------------------------------
        // 6. Clear the interrupt flag and re‑enable ranging
        // --------------------------------------------------------------------
        // The DW1000 automatically clears its interrupt status when we read
        // the registers, but we can explicitly re‑arm the receiver.
        // The library's handleRanging() method will process the next frame.
        // However, we are not using the full ranging loop – we just read diags.
        // We simply restart receiver listening.
        DW1000Ranging.startRanging(); // re‑start listening for new frames

        // If you want to keep the receiver always on, you can call
        // DW1000Ranging.forceRangingOff(); // but that's not needed here.
    }

    // ------------------------------------------------------------------------
    // 7. Background tasks (optional)
    // ------------------------------------------------------------------------
    // If we were using the library's ranging state machine, we'd call
    // DW1000Ranging.loop() here. But since we only read diagnostics on
    // interrupt, we don't need it. However, to keep the library happy,
    // we can call it occasionally (but it might interfere).
    // Instead, we just yield to the system.
    delay(1); // small delay to avoid watchdog issues
}
